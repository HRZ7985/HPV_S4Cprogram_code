# Define the function
extract_fixed1 <- function(model, model_name) {
  coef_sum <- summary(model)$coefficients
  OR <- exp(coef_sum[, "Estimate"])
  CI_low <- exp(coef_sum[, "Estimate"] - 1.96 * coef_sum[, "Std. Error"])
  CI_up <- exp(coef_sum[, "Estimate"] + 1.96 * coef_sum[, "Std. Error"])
  p_val <- coef_sum[, "Pr(>|z|)"]
  
  result <- data.frame(
    Model = model_name,
    Variable = rownames(coef_sum),
    OR = sprintf("%.2f", OR),
    CI = sprintf("(%.2f, %.2f)", CI_low, CI_up),
    P_value = sprintf("%.3f", p_val),
    stringsAsFactors = FALSE
  )
  result$Variable[result$Variable == "(Intercept)"] <- "Intercept"
  return(result)
}

# Define an extraction function
extract_fixed2 <- function(model, model_name) {
  coef_sum <- summary(model)$coefficients
  est <- coef_sum[, "Estimate"]
  se <- coef_sum[, "Std. Error"]
  CI_low <- est - 1.96 * se
  CI_up <- est + 1.96 * se
  p_val <- coef_sum[, "Pr(>|t|)"]
  
  result <- data.frame(
    Model = model_name,
    Variable = rownames(coef_sum),
    Estimate = sprintf("%.2f", est),
    CI = sprintf("(%.2f, %.2f)", CI_low, CI_up),
    P_value = ifelse(p_val < 0.001, "<0.001", sprintf("%.3f", p_val)),
    stringsAsFactors = FALSE
  )
  return(result)
}

##Effect size output
extract_fixed3 <- function(model, model_name) {
  s <- summary(model)
  coefs <- s$coefficients
  OR <- exp(coefs[, 1])
  lwr <- exp(coefs[, 1] - 1.96 * coefs[, 2])
  upr <- exp(coefs[, 1] + 1.96 * coefs[, 2])
  or_ci <- sprintf("%.2f (%.2f, %.2f)", OR, lwr, upr)
  p_val <- coefs[, 4]
  p_formatted <- ifelse(p_val < 0.001, "<0.001", sprintf("%.3f", p_val))
  
  result <- data.frame(
    Model = model_name,
    Term = rownames(coefs),
    `OR (95% CI)` = or_ci,
    p.value = p_formatted,
    check.names = FALSE, 
    stringsAsFactors = FALSE
  )
  return(result)
}

# Effect size output
extract_fixed4 <- function(model, model_name) {
  s <- summary(model)
  coefs <- s$coefficients 
  beta <- coefs[, 1]                
  se   <- coefs[, 2]                
  lwr  <- beta - 1.96 * se         
  upr  <- beta + 1.96 * se          
  beta_ci <- sprintf("%.2f (%.2f, %.2f)", beta, lwr, upr)
  p_val <- coefs[, 4]
  p_formatted <- ifelse(p_val < 0.001, "<0.001", sprintf("%.3f", p_val))
  
  result <- data.frame(
    Model = model_name,
    Term = rownames(coefs),
    `β (95% CI)` = beta_ci,
    p.value = p_formatted,
    check.names = FALSE,
    stringsAsFactors = FALSE
  )
  return(result)
}
